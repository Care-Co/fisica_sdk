cmake_minimum_required(VERSION 3.16)
project(FisicaPartnerSamples LANGUAGES C CXX)
set(FISICA_SDK_DIR "" CACHE PATH "Extracted public core bundle; never the private source directory")
if(NOT EXISTS "${FISICA_SDK_DIR}/fisica_core.h")
  message(FATAL_ERROR "Set FISICA_SDK_DIR to the extracted core bundle")
endif()
add_library(fisica_core SHARED IMPORTED)
if(WIN32)
  if(NOT MSVC)
    message(FATAL_ERROR "Windows sample uses MSVC; choose a matching x64/ARM64 developer environment")
  endif()
  if(CMAKE_GENERATOR_PLATFORM STREQUAL "ARM64" OR CMAKE_SYSTEM_PROCESSOR MATCHES "^(ARM64|aarch64)$")
    set(FISICA_MACHINE ARM64)
  else()
    set(FISICA_MACHINE X64)
  endif()
  # Generate an import library from public exports, without compiling private code.
  execute_process(COMMAND "${CMAKE_AR}" /DEF:${CMAKE_CURRENT_SOURCE_DIR}/c/fisica_core.def /MACHINE:${FISICA_MACHINE} /OUT:${CMAKE_CURRENT_BINARY_DIR}/fisica_core.lib RESULT_VARIABLE IMPORT_RESULT)
  if(NOT IMPORT_RESULT EQUAL 0)
    message(FATAL_ERROR "Could not generate the core import library")
  endif()
  set_target_properties(fisica_core PROPERTIES IMPORTED_LOCATION "${FISICA_SDK_DIR}/fisica_core.dll" IMPORTED_IMPLIB "${CMAKE_CURRENT_BINARY_DIR}/fisica_core.lib")
elseif(APPLE)
  set_target_properties(fisica_core PROPERTIES IMPORTED_LOCATION "${FISICA_SDK_DIR}/libfisica_core.dylib")
else()
  # Linux cores are built for SDK CI checks only; no Linux package is supplied.
  set_target_properties(fisica_core PROPERTIES IMPORTED_LOCATION "${FISICA_SDK_DIR}/libfisica_core.so" IMPORTED_NO_SONAME TRUE)
endif()
foreach(language c cpp)
  add_executable(fisica-${language} ${language}/main.${language})
  target_include_directories(fisica-${language} PRIVATE "${FISICA_SDK_DIR}")
  target_link_libraries(fisica-${language} PRIVATE fisica_core)
  if(WIN32)
    add_custom_command(TARGET fisica-${language} POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${FISICA_SDK_DIR}/fisica_core.dll" "$<TARGET_FILE_DIR:fisica-${language}>")
  elseif(APPLE)
    set_target_properties(fisica-${language} PROPERTIES BUILD_RPATH "@loader_path")
    add_custom_command(TARGET fisica-${language} POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${FISICA_SDK_DIR}/libfisica_core.dylib" "$<TARGET_FILE_DIR:fisica-${language}>")
  else()
    set_target_properties(fisica-${language} PROPERTIES BUILD_RPATH "$ORIGIN")
    add_custom_command(TARGET fisica-${language} POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${FISICA_SDK_DIR}/libfisica_core.so" "$<TARGET_FILE_DIR:fisica-${language}>")
  endif()
endforeach()
